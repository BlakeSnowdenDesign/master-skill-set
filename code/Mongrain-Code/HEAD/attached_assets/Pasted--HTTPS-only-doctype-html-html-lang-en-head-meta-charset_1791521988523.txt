    if(e.target.closest('.ov,.lab,.callout'))return;
    dragging=true;moved=false;ptype=e.pointerType;sx=e.clientX;sy=e.clientY;syaw=yaw;spitch=pitch;
  });
  window.addEventListener('pointermove',function(e){
    if(!dragging)return;var dx=e.clientX-sx,dy=e.clientY-sy;
    if(!moved){if(Math.abs(dx)+Math.abs(dy)<5)return;if(ptype==='touch'&&Math.abs(dy)>Math.abs(dx)){dragging=false;return;}
      moved=true;stage.classList.add('drag');co.classList.remove('show');hoverId=null;lit();hint.classList.add('hide');}
    yaw=Math.max(-82,Math.min(14,syaw+dx*.25));if(ptype!=='touch')pitch=Math.max(-58,Math.min(-4,spitch-dy*.2));
    setCam(0);
  });
  window.addEventListener('pointerup',function(){if(!dragging)return;dragging=false;stage.classList.remove('drag');setTimeout(function(){moved=false;},0);});
  window.addEventListener('pointercancel',function(){dragging=false;stage.classList.remove('drag');});
  stage.addEventListener('click',function(e){if(moved||e.target.closest('.ov,.lab,.layer'))return;if(lockId||catId)unlock();});

  /* ---------- Controls ---------- */
  root.querySelectorAll('.seg button').forEach(function(b){b.addEventListener('click',function(){clearSeq();setT(+b.dataset.view);idle=true;setCam(1100);});});
  sep.addEventListener('input',function(){clearSeq();model.classList.add('live');setT(sep.value/100,{live:true});setCam(0);});
  sep.addEventListener('change',function(){model.classList.remove('live');setCam(500);});
  $('#b-play').addEventListener('click',play);
  $('#b-reset').addEventListener('click',function(){clearSeq();scope=null;uz=1;syncGroups();setT(0);yaw=Y0;pitch=P0;sway=0;tIdle=0;idle=true;unlock();now.innerHTML='<b>Assembled</b> · hover a layer, or switch to Exploded';setCam(1200);});
  $('#z-in').addEventListener('click',function(){uz=Math.min(3,uz*1.25);setCam(500);});
  $('#z-out').addEventListener('click',function(){uz=Math.max(.6,uz/1.25);setCam(500);});
  $('#z-fit').addEventListener('click',function(){uz=1;yaw=Y0;pitch=P0;sway=0;setCam(800);});
  $('#prev').addEventListener('click',function(){step(-1);});$('#next').addEventListener('click',function(){step(1);});
  stage.addEventListener('keydown',function(e){
    if(e.key==='ArrowRight'||e.key==='ArrowDown'){e.preventDefault();step(1);}
    else if(e.key==='ArrowLeft'||e.key==='ArrowUp'){e.preventDefault();step(-1);}
    else if(e.key==='Escape'){unlock();}
    else if(e.key==='e'||e.key==='E'){setT(T>.5?0:1);setCam(1100);}
  });

  /* ---------- Start ---------- */
  var started=false;
  function go(){root.classList.add('in');if(started)return;started=true;
    if(PLAY_ON_LOAD&&!reduce)setTimeout(play,450);else{setT(0,{instant:true});idle=true;setCam(0);now.innerHTML='<b>Assembled</b> · hover a layer, or switch to Exploded';}}
  syncGroups();setT(0,{instant:true});setCam(0);defDetail();phases.classList.add('hide');
  if(PLAY_ON_LOAD&&!reduce)L.forEach(function(l){l.el.classList.add('gone');});
  requestAnimationFrame(loop);
  setTimeout(function(){hint.classList.add('hide');},14000);
  if(reduce||!('IntersectionObserver' in window))go();
  else new IntersectionObserver(function(es){if(es[0].isIntersecting)go();},{threshold:.2}).observe(stage);
  setTimeout(go,2500);
  var rT;window.addEventListener('resize',function(){clearTimeout(rT);rT=setTimeout(function(){syncSeg();setCam(0);},120);});
  if(document.fonts&&document.fonts.ready)document.fonts.ready.then(syncSeg);
})();
